MATCH (f:Fixture)<-[:PLAYED_IN]-(p:Player)-[:PLAYS_AS]->(pos1:Position)
WHERE EXISTS {
    MATCH (p)-[:PLAYS_AS]->(pos2:Position)
    WHERE pos2.name <> pos1.name
}

WITH p, COUNT(DISTINCT f) AS total_fixtures

MATCH (f:Fixture)<-[pi:PLAYED_IN]-(p)
WITH p, total_fixtures, f, pi,
toFloat(
  (pi.assists * 3)
  + CASE WHEN pi.minutes > 0 THEN 1 ELSE 0 END
  + CASE WHEN pi.minutes >= 60 THEN 1 ELSE 0 END
  - (pi.penalties_missed * 2)
  - (pi.own_goals * 2)
  - (pi.yellow_cards)
  - (pi.red_cards * 3)
  + (pi.goals_scored * 6)
  + (pi.clean_sheets * 4 * CASE WHEN pi.minutes > 0 THEN 1 ELSE 0 END)
  - toInteger(pi.goals_conceded / 2)
  + (pi.bonus)
) AS def_score

WITH p, total_fixtures,
     COLLECT(
       CASE WHEN pi.total_points <> def_score
       THEN {
         def_score: def_score,
         season: f.season,
         fixture_number: f.fixture_number,
         matches_def: false,
         total_points: pi.total_points
       } END
     ) AS fixture_details

RETURN p.player_name AS player,
       SIZE([x IN fixture_details WHERE x IS NOT NULL]) AS not_def_fixtures,
       total_fixtures,
       [x IN fixture_details WHERE x IS NOT NULL] AS fixture_details
